cmake_minimum_required(VERSION 3.20.0)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(meshcore_cayenne_lpp_compat_tdd_test)

include(${CMAKE_CURRENT_LIST_DIR}/../../common/reference_meshcore.cmake)

zephyr_include_directories(
  ${MESHCORE_INTERNAL_INCLUDE_DIRS}
  ${MESHCORE_TEST_COMMON_DIR}/lib/CayenneLpp
)

target_sources(app PRIVATE
  src/main.cpp
  src/reference_polyline_stub.cpp
  ${MESHCORE_SUPPORT_CAYENNE_LPP_SOURCES}
  ${MESHCORE_TEST_COMMON_DIR}/lib/CayenneLpp/CayenneLPP.cpp
)

target_compile_definitions(app PRIVATE
  CAYENNE_DISABLE_DIGITAL_INPUT
  CAYENNE_DISABLE_DIGITAL_OUTPUT
  CAYENNE_DISABLE_ANALOG_OUTPUT
  CAYENNE_DISABLE_GENERIC_SENSOR
  CAYENNE_DISABLE_LUMINOSITY
  CAYENNE_DISABLE_PRESENCE
  CAYENNE_DISABLE_ACCELEROMETER
  CAYENNE_DISABLE_FREQUENCY
  CAYENNE_DISABLE_PERCENTAGE
  CAYENNE_DISABLE_ENERGY
  CAYENNE_DISABLE_DIRECTION
  CAYENNE_DISABLE_UNIX_TIME
  CAYENNE_DISABLE_GYROMETER
  CAYENNE_DISABLE_SWITCH
  CAYENNE_DISABLE_CONCENTRATION
  CAYENNE_DISABLE_COLOUR
)

zephyr_library_compile_options(
  -Wno-unused-parameter
  -Wno-unused-variable
  $<$<COMPILE_LANGUAGE:CXX>:-Wno-reorder>
)
